Phase 1: Intent Classification (Token Gate)

Goal: Avoid sending expensive Claude Opus calls for questions that can be answered for free (Genie Space or FAQ cache).

Saves: 40-60% of requests from hitting Opus

Implementation

# intent_classifier.py — Lightweight classification (runs BEFORE any LLM call)

import re

# Pattern-based classifier (zero tokens)
_DATA_PATTERNS = [
    r'\b(show|list|count|how many|get|find|select|query|fetch)\b',
    r'\b(table|column|row|record|data|job|pipeline|run|migration)\b',
    r'\b(last|recent|today|yesterday|this week|failed|success)\b',
    r'\b(average|sum|total|max|min|group by|order by)\b',
]

_HOWTO_PATTERNS = [
    r'\b(how to|how do i|what is|explain|help|guide|steps|tutorial)\b',
    r'\b(configure|setup|install|create|build|deploy|connect)\b',
]

def classify_intent(question: str) -> str:
    """Classify user intent without any LLM call.
    Returns: 'data_query' | 'how_to' | 'general'
    """
    q = question.lower().strip()

    # Check FAQ cache first (instant, zero tokens)
    if _matches_faq(q):
        return 'faq'

    # Score data-query likelihood
    data_score = sum(1 for p in _DATA_PATTERNS if re.search(p, q))
    howto_score = sum(1 for p in _HOWTO_PATTERNS if re.search(p, q))

    if data_score >= 2:
        return 'data_query'  # Route to Genie Space (0 tokens)
    elif howto_score >= 2:
        return 'how_to'      # Route to Claude with SLIM context
    else:
        return 'general'     # Route to Claude with FULL context

def _matches_faq(q: str) -> bool:
    """Check if question matches pre-cached FAQ patterns."""
    # Reuse existing FAQ from genie.py
    from routes.genie import _check_faq
    return _check_faq(q) is not None

Phase 3: Conversation History Windowing

Goal: Reduce history from 10 messages to a smart sliding window.

Saves: 200-600 tokens per multi-turn conversation

Strategy: Recency + Relevance Hybrid

def _compress_history(messages: list, current_question: str, max_tokens: int = 500) -> list:
    """
    Smart history compression:
    1. Always keep last 2 messages (immediate context)
    2. Summarise older messages into a single system note
    3. Drop messages that exceed token budget
    """
    if len(messages) <= 2:
        return messages

    recent = messages[-2:]         # ~200 tokens (keep verbatim)
    older = messages[:-2]          # Candidates for compression

    # Summarise older context into 1 line
    topics = set()
    for msg in older:
        content = msg.get("content", "").lower()
        if "sql" in content or "query" in content:
            topics.add("SQL queries")
        if "migration" in content:
            topics.add("migration")
        if "pipeline" in content:
            topics.add("pipelines")

    summary = f"[Prior conversation covered: {', '.join(topics) or 'general discussion'}]"
    compressed = [{"role": "system", "content": summary}] + recent
    return compressed  # ~250 tokens instead of ~800

Phase 4: Response Caching (Semantic Deduplication)

Goal: Cache responses for similar/repeated queries. Avoid re-calling Claude for the same question.

Saves: 100% tokens on cache hits (~30% hit rate expected)

Implementation: LRU Cache with Fuzzy Matching

import hashlib
from collections import OrderedDict

class ResponseCache:
    """LRU cache for FM responses. Fuzzy-matches similar questions."""

    def __init__(self, max_size: int = 100, ttl_seconds: int = 3600):
        self._cache = OrderedDict()  # key -> {response, timestamp}
        self._max_size = max_size
        self._ttl = ttl_seconds

    def _normalise(self, question: str) -> str:
        """Normalise question for fuzzy matching."""
        q = question.lower().strip()
        # Remove filler words
        for w in ['please', 'can you', 'show me', 'i want to', 'give me']:
            q = q.replace(w, '')
        # Normalise whitespace
        q = ' '.join(q.split())
        return q

    def _key(self, question: str) -> str:
        normalised = self._normalise(question)
        return hashlib.md5(normalised.encode()).hexdigest()

    def get(self, question: str) -> dict | None:
        key = self._key(question)
        if key in self._cache:
            entry = self._cache[key]
            age = time.time() - entry['timestamp']
            if age < self._ttl:
                self._cache.move_to_end(key)  # LRU refresh
                return entry['response']
            else:
                del self._cache[key]  # Expired
        return None

    def put(self, question: str, response: dict):
        key = self._key(question)
        self._cache[key] = {'response': response, 'timestamp': time.time()}
        if len(self._cache) > self._max_size:
            self._cache.popitem(last=False)  # Evict oldest

# Global instance
_fm_cache = ResponseCache(max_size=200, ttl_seconds=1800)  # 30 min TTL

        for chunk in raw.iter_lines():
            if chunk:
                # Parse SSE chunk, extract delta content
                data = json.loads(chunk.decode())
                delta = data["choices"][0]["delta"].get("content", "")
                if delta:
                    yield f"data: {json.dumps({'text': delta})}\n\n"
        yield "data: [DONE]\n\n"

    return Response(stream_with_context(generate()),
                    mimetype='text/event-stream')

function streamFmChat(bid, question) {

    let fullText = '';
    es.onmessage = function(e) {
        if (e.data === '[DONE]') { es.close(); return; }
        const chunk = JSON.parse(e.data);
        fullText += chunk.text;
        updateUI(bid, fullText);  // Real-time render
    };
    // User navigates away → abort (saves output tokens)
    window.addEventListener('beforeunload', () => es.close());
}

Phase 7: Token Budget Controller

Goal: Enforce per-user, per-session, and per-day token limits.

Prevents runaway costs from loops or abuse

Implementation

class TokenBudget:
    """Per-user token budget with soft/hard limits."""

    LIMITS = {
        'per_request': 4096,       # Max output tokens per request
        'per_session': 50_000,     # Soft limit per session (warn user)
        'per_day': 200_000,        # Hard limit per user per day
    }

    def __init__(self):
        self._usage = {}  # user_email -> {today: int, session: int}

    def check(self, user_email: str, estimated_tokens: int) -> tuple[bool, str]:
        usage = self._usage.setdefault(user_email, {'today': 0, 'session': 0})

        if usage['today'] + estimated_tokens > self.LIMITS['per_day']:
            return False, "Daily token limit reached. Resets at midnight."
        if usage['session'] > self.LIMITS['per_session']:
            return True, "⚠️ High token usage this session. Consider shorter questions."
        return True, ""

    def record(self, user_email: str, tokens_used: int):
        usage = self._usage.setdefault(user_email, {'today': 0, 'session': 0})
        usage['today'] += tokens_used
        usage['session'] += tokens_used

Key Principle

The fastest, cheapest token is the one you never send. Route data queries to the Genie Space (0 tokens), cache repeated answers (0 tokens), and only invoke Claude when genuine reasoning is needed — using the cheapest model that can handle the complexity.
